# =====================================================================
# 200 m, ~20 m/s, ~20 Hz, nadir RGB camera: the planar frontend
# (see README "High altitude: the planar frontend")
# Replace <OFFSET> with the camera clock offset from step 0a.
# =====================================================================

# 0a. camera clock offset
python tools/estimate_time_offset.py --dataset data --output artifacts/time_offset.json

# 0b. how far the ground moves per frame gap -> pick --frame-gap / --max-frame-gap-s
python tools/check_motion_budget.py --dataset data --calibration configs/vo/camera_fixedwing.json --image-time-offset <OFFSET> --output artifacts/motion_budget.json

# 0c. which way the camera points; paste the printed "mounting" block into
#     configs/vo/camera_fixedwing.json (or pass --camera-mounting artifacts/camera_mounting.json).
#     "measured / predicted motion" must be close to 1.0 before training.
python tools/estimate_camera_mounting.py --dataset data --calibration configs/vo/camera_fixedwing.json --image-time-offset <OFFSET> --output artifacts/camera_mounting.json

# 1. split with the gap training will use (20 frames = 1 s at 20 Hz)
python tools/split_dataset.py --dataset data --output-root data_split --image-time-offset <OFFSET> --frame-gap 20 --deployment-latency-s 0.35 --link

# 2. floor to beat: the UNTRAINED geometric estimate (learning rate 0, one epoch)
python tools/train_fixedwing_vo.py --dataset data_split/train --validation-dataset data_split/validation --test-dataset data_split/test --calibration configs/vo/camera_fixedwing.json --camera-mounting artifacts/camera_mounting.json --frontend planar --color --frame-gap 20 --max-frame-gap-s 1.5 --warmup 140 --correlation-radius 3 --image-time-offset <OFFSET> --velocity-loss simple --learning-rate 0 --epochs 1 --run-dir runs/planar_untrained --device 0,1 --num-workers 2 --frontend-chunk 4
