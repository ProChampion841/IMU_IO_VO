# IO (velnet) -- commands.  Run everything from the IO/ folder.
#
# Current model: configs/exp/UAV/velnet_v3.conf  (updated 2026-10-02)
#   input   acc (m/s^2, body FLU), gyro (rad/s, body FLU), airspeed (m/s, pitot), 100 Hz
#   output  vel_body (m/s, body FLU) = (airspeed, 0, 0) + learned correction
#           vel_cov  per-axis variance, (m/s)^2
# Checkpoints from before 2026-10-02 do NOT load into velnet_v3 (airspeed adds an
# input channel).  Start a fresh run.  Older runs: use their own config.


# ===== SETUP (once) =====
pip install torch pypose pyhocon pandas scipy tqdm matplotlib tensorboard onnx onnxruntime


# ===== TRAIN =====
python train.py --config configs/exp/UAV/velnet_v3.conf --device cuda:0

# resume after a stop
python train.py --config configs/exp/UAV/velnet_v3.conf --device cuda:0 --load_ckpt

# watch training (all runs side by side)
tensorboard --logdir experiments/UAV --host 0.0.0.0

# results: experiments/UAV/velnet_v3/metric.csv, parameters.yaml,
#          ckpt/best_model.ckpt (selected on val_vel_error_60s), ckpt/0000.ckpt ... (every epoch)
# There is no early stopping: stop the run by hand once val_vel_error_60s stops improving.


# ===== EVALUATE: 30s 1min 2min 5min 10min 15min 20min 30min 40min =====
# eval      = validation flights (12)
# inference = held-out test flights (11), never used in training or selection
# vel/dir errors pooled over every frame; pos_error = drift at the END of each window.
# No GPU: add  --device cpu

# 1) which flights are long enough for each horizon (no checkpoint needed)
python -m tools.eval_vel_horizons --config configs/exp/UAV/velnet_v3.conf --splits eval inference --check

# 2) all horizons, validation + test flights
python -m tools.eval_vel_horizons --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --splits eval inference --out experiments/UAV/velnet_v3/horizons.csv

# 3) same, with the first-frame velocity offset removed (world frame)
python -m tools.eval_vel_horizons --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --splits eval inference --v0_offset --out experiments/UAV/velnet_v3/horizons_v0.csv

# 4) one flight (any unique part of the file name)
python -m tools.eval_vel_horizons --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --splits eval --flights 178_47 --out experiments/UAV/velnet_v3/horizons_178_47.csv

# 5) a chosen epoch
python -m tools.eval_vel_horizons --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/0003.ckpt --splits eval inference --out experiments/UAV/velnet_v3/horizons_ep3.csv

# 6) your own horizons, in FRAMES at 100 Hz (here 30 s, 1, 2, 3, 4 min)
python -m tools.eval_vel_horizons --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --splits eval inference --horizons 3000 6000 12000 18000 24000 --out experiments/UAV/velnet_v3/horizons_short.csv


# ===== PLOT =====
# window mode: every 60 s window starts from the true state (as in training)
python -m tools.plot_trajectory --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --split eval --seq 178_47 --length 6000

# continuous mode: one unbroken trajectory, true state used only at the start
python -m tools.plot_trajectory --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --split eval --seq 178_47 --length 6000 --mode continuous

# any CSV file directly (add --no_raw to skip the raw-IMU comparison line)
python -m tools.plot_trajectory --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --csv /path/to/flight_sensor_data.csv --length 6000


# ===== ONNX (deployment, no PyTorch needed at run time) =====
# export; the window length is FIXED (6000 = 60 s); inputs acc, gyro, airspeed
python -m tools.export_onnx --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --out experiments/UAV/velnet_v3/velnet_6000.onnx --frames 6000

# run on a flight CSV (raw log with AirSpeed; GPS columns, if present, are used for scoring)
python -m tools.onnx_inference --onnx experiments/UAV/velnet_v3/velnet_6000.onnx --csv /path/to/flight_sensor_data.csv --out pred.csv

# check the ONNX path against the PyTorch pipeline (synthetic flight if no --csv)
python -m tools.check_onnx_inference --config configs/exp/UAV/velnet_v3.conf --ckpt experiments/UAV/velnet_v3/ckpt/best_model.ckpt --onnx experiments/UAV/velnet_v3/velnet_6000.onnx
