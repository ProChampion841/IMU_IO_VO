:: ============================================================================
:: EKF (IMU + VO) commands  (Windows cmd;  ^ = line continues)
:: Run from the EKF folder.  EKF\ and IMU\ must sit side by side.
:: Horizons are frames at 100 Hz: 3000 = 30 s, 6000 = 1 min ... 30000 = 5 min.
:: The window is the LONGEST horizon; every shorter one is read off its prefix.
:: ============================================================================

:: ---------------------------------------------------------------- GPS-FREE (2026-10-02)
:: Whole flight with no GPS: MTi tilt (configs\ekf_gpsfree.json), land matching, learned IMU
:: model trained GPS-free.  Train / export steps: ..\IMU\cmd.txt, section "GPS-FREE".
python run_stream.py --imu_config ../IMU/configs/exp/UAV/gpsfree_rotate_smooth.conf --splits inference --vo_dir vo_cache --ekf_config configs/ekf_gpsfree.json --no_gt --pos_sim --imu model --imu_onnx ../IMU/gpsfree_40s.onnx --horizons 30s 1m 2m 5m 10m --out_csv gpsfree_stream_model.csv

:: ---------------------------------------------------------------- tests
python -m pytest tests -q

:: ---------------------------------------------------------------- simulated VO
:: Checks the filter on your REAL IMU flights with SIMULATED VO (GPS truth + noise
:: sized from the measured VO RMSE).  Not a VO result -- the output says so.
python run_ekf.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_sim --horizons 3000 6000 12000 --out_csv ekf_sim.csv

:: ---------------------------------------------------------------- VO ONNX model
:: Runs VO\tools\onnx_inference.py on the VO flight folder (flight.csv + images\),
:: keeps one output per delivered image pair, uses VO's own variance.
:: Export first:  cd ..\VO  &  python tools\export_onnx.py runs\vo_planar_s0\best.pt --output-dir export\onnx
python run_ekf.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --csv 2026_02_06_143_23_sensor_data.csv ^
  --vo_onnx ..\VO\export\onnx --vo_dataset ..\VO\data_split\test ^
  --vo_cache_dir vo_cache --horizons 3000 6000 12000 --plot_dir ekf_plots

:: many flights: vo_flights\<flight name>\flight.csv + images\ per IMU flight
python run_ekf.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_onnx ..\VO\export\onnx --vo_datasets vo_flights ^
  --vo_cache_dir vo_cache --horizons 3000 6000 12000 18000 24000 30000 --out_csv ekf_windows.csv

:: ---------------------------------------------------------------- VO CSV
:: vo_predictions\<flight>_vo.csv per flight: time, body_velocity_m_s_x/y/z (FRD),
:: velocity_log_variance_x/y/z -- the VO model's output contract.
python run_ekf.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_dir vo_predictions ^
  --horizons 3000 6000 12000 18000 24000 30000 ^
  --out_csv ekf_windows.csv --out_npz ekf_traj.npz --plot_dir ekf_plots

:: same, with the learned IMU correction (export it first, see IMU\cmd.txt)
python run_ekf.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_dir vo_predictions --imu_onnx ..\IMU\tilt_rotate_40s.onnx ^
  --horizons 3000 6000 12000 18000 24000 30000 --out_csv ekf_windows_onnx.csv

:: one flight
python run_ekf.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --csv 2026_02_06_143_23_sensor_data.csv ^
  --vo_csv vo_predictions\2026_02_06_143_23_sensor_data_vo.csv ^
  --horizons 3000 6000 12000 --plot_dir ekf_plots

:: settings (noise, attitude aid, VO variance scale): configs\ekf_default.json
:: copy it, edit, and pass --ekf_config my.json

:: ---------------------------------------------------------------- STREAM (real use)
:: GPS for 60 s (biases learned), then outage: EKF vs IMU-only at 30 s .. 5 min
python run_stream.py --csv 2026_02_06_143_23_sensor_data.csv ^
  --vo_onnx ..\VO\export\onnx --vo_dataset ..\VO\data_split\test --vo_cache vo_143_23.csv ^
  --gps_s 60 --horizons 30s 1m 2m 3m 4m 5m

:: ---------------------------------------------------------------- LAND MATCHING (absolute position)
:: SIMULATED fixes from GPS truth (10 m, 1 Hz, 0.5 s late; settings: pos_sim in the config)
python run_stream.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_dir vo_cache --pos_sim --gps_s 60 ^
  --horizons 30s 1m 2m 5m 10m --out_csv stream_landmatch.csv
:: your matcher's fixes: landmatch\<flight>_pos.csv (time, lat, lon[, std][, arrival]),
:: origin = GPS position at the log start.  VO is optional.
python run_stream.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --pos_dir landmatch --pos_origin 37.123456 127.123456 50.0 ^
  --gps_s 60 --horizons 30s 1m 2m 5m 10m
:: check C++ = Python on a log with late fixes
python run_stream.py --csv 2026_02_06_143_23_sensor_data.csv --vo_csv vo_143_23.csv ^
  --pos_sim --events_out events_pos.csv --cpp cpp\build\ekf_replay

:: ---------------------------------------------------------------- C++ (Jetson)
cd cpp
cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
cmake --build build -j
cd ..
:: run the C++ on the same message log as Python and compare (prints PASS/FAIL)
python run_stream.py --csv 2026_02_06_143_23_sensor_data.csv --vo_csv vo_143_23.csv ^
  --events_out events.csv --cpp cpp\build\ekf_replay

:: ---------------------------------------------------------------- ALL HORIZONS 30s .. 40m
:: horizons can be written 30s 1m 2m 3m 4m 5m 10m 15m 20m 30m 40m (that list is the default)
:: OFFLINE: each horizon on its own windows; --step 1m lets long horizons fit shorter flights
python run_ekf.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_dir vo_cache --per_horizon --step 1m ^
  --horizons 30s 1m 2m 3m 4m 5m 10m 15m 20m 30m 40m --out_csv ekf_all_horizons.csv

:: ONLINE / STREAM: every flight of the split, GPS 60 s then outage, horizons into the outage
python run_stream.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_dir vo_cache --gps_s 60 ^
  --horizons 30s 1m 2m 3m 4m 5m 10m 15m 20m 30m 40m --out_csv stream_all_horizons.csv

:: ---------------------------------------------------------------- NO GROUND TRUTH, NO RESET
:: One continuous run per flight: start from the first VO + nav attitude (position 0),
:: no GPS, no reset.  GPS truth is only used to score it.  Horizons = time since start.
python run_stream.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_dir vo_cache --no_gt ^
  --horizons 30s 1m 2m 3m 4m 5m 10m 15m 20m 30m 40m --out_csv no_gt.csv

:: same, VO straight from the ONNX model, and check the C++ build on the first flight
python run_stream.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_onnx ..\VO\export\onnx --vo_datasets vo_flights --vo_cache_dir vo_cache ^
  --no_gt --events_out events_nogt.csv --cpp cpp\build\ekf_replay

python run_stream.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^

:: learned IMU model
python run_stream.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_dir vo_cache --no_gt --imu model --imu_onnx ..\IMU\tilt_rotate_40s.onnx ^
  --out_csv no_gt_model.csv
:: with GPS 60 s first: the model starts at the outage, with the 15 s freeze it was trained with
python run_stream.py --imu_config ..\IMU\configs\exp\UAV\tilt_rotate.conf ^
  --splits inference --vo_dir vo_cache --gps_s 60 --imu model --imu_onnx ..\IMU\tilt_rotate_40s.onnx
