:: ============================================================================
:: AirIMU commands  (Windows cmd;  ^ = line continues)
:: Run from the repo folder, in the mvvio env:   conda activate mvvio
::
:: Splits:  inference = HELD-OUT flights (use this for real results)
::          test / eval = the flights used to PICK the checkpoint (not held-out)
:: Horizons are in frames at 100 Hz:  3000 = 30 s,  6000 = 60 s,  12000 = 120 s
:: ============================================================================


:: ############################################################################
:: GPS-FREE: TRAIN AND EVALUATE  (2026-10-02)  -- START HERE
:: Model: configs/exp/UAV/gpsfree_rotate_smooth.conf
::   = tilt_rotate + smooth tilt-correction loss, with NO GPS-aided input:
::     MTi attitude for the network input AND for gravity removal, no GPS bias freeze.
:: Every command is ONE line (works in Windows cmd and in a Linux shell).
:: ############################################################################

:: ---- 1. TRAIN  (IMU folder)
python train.py --config configs/exp/UAV/gpsfree_rotate_smooth.conf --device cuda:0

:: resume a stopped run from its newest checkpoint
python train.py --config configs/exp/UAV/gpsfree_rotate_smooth.conf --device cuda:0 --load_ckpt

:: second seed, to check the result is not luck
python train.py --config configs/exp/UAV/gpsfree_rotate_smooth.conf --device cuda:0 --seed 1

:: watch live (http://localhost:6006): val_60s/pos_error_m vs val_60s/raw_pos_error_m,
:: and val/dtheta_step_rms_deg (how jumpy the tilt correction is)
tensorboard --logdir experiments/UAV/gpsfree_rotate_smooth/tensorboard --host 0.0.0.0

:: ---- 2. EVALUATE on the HELD-OUT flights (PyTorch checkpoint)
:: per flight, 30 / 60 / 120 s, model vs raw IMU (raw = same MTi attitude, no freeze)
python -m tools.eval_vel_horizons1 --config configs/exp/UAV/gpsfree_rotate_smooth.conf --ckpt experiments/UAV/gpsfree_rotate_smooth/ckpt/best_model.ckpt --splits inference --nested --horizons 3000 6000 12000 --per_flight --out gpsfree_heldout.csv

:: ---- 3. EXPORT TO ONNX and check it on the held-out flights
:: inputs acc, gyro, g_body (B, 4009, 3); on the drone g_body comes from the MTi attitude
python -m tools.export_onnx --config configs/exp/UAV/gpsfree_rotate_smooth.conf --ckpt experiments/UAV/gpsfree_rotate_smooth/ckpt/best_model.ckpt --frames 4000 --out gpsfree_40s.onnx

:: same numbers from the ONNX file (should match step 2 at 30 / 40 s), and ONNX = PyTorch check
python -m tools.onnx_inference --onnx gpsfree_40s.onnx --config configs/exp/UAV/gpsfree_rotate_smooth.conf --splits inference --horizons 3000 4000 --compare_ckpt experiments/UAV/gpsfree_rotate_smooth/ckpt/best_model.ckpt --out_csv gpsfree_onnx_windows.csv

:: ---- 4. WHOLE-FLIGHT TEST WITHOUT ANY GPS  (EKF folder: cd ..\EKF)
:: no GPS at all: start from VO + MTi attitude at position 0, one run per flight, GPS only
:: scores it.  EKF uses the MTi tilt (configs/ekf_gpsfree.json) and simulated land-matching
:: fixes (--pos_sim; replace with --pos_dir landmatch --pos_origin LAT LON ALT for real ones).
:: (a) learned IMU model
python run_stream.py --imu_config ../IMU/configs/exp/UAV/gpsfree_rotate_smooth.conf --splits inference --vo_dir vo_cache --ekf_config configs/ekf_gpsfree.json --no_gt --pos_sim --imu model --imu_onnx ../IMU/gpsfree_40s.onnx --horizons 30s 1m 2m 5m 10m --out_csv gpsfree_stream_model.csv
:: (b) same, raw IMU -- the model must beat this
python run_stream.py --imu_config ../IMU/configs/exp/UAV/gpsfree_rotate_smooth.conf --splits inference --vo_dir vo_cache --ekf_config configs/ekf_gpsfree.json --no_gt --pos_sim --imu raw --horizons 30s 1m 2m 5m 10m --out_csv gpsfree_stream_raw.csv

:: NOTE  compare GPS-free results only with GPS-free results: the raw baseline also
:: changed (MTi attitude, no freeze), so ratios are not comparable with tilt_rotate.
:: ############################################################################


:: ---------------------------------------------------------------- TRAIN

python train.py --config configs/exp/UAV/tilt_aware.conf --device cuda:0

python train.py --config configs/exp/UAV/tilt_aware.conf --device cuda:0 --load_ckpt

python -m tools.eval_vel_horizons --config configs/exp/UAV/tilt_aware.conf ^
  --ckpt experiments/UAV/tilt_aware/ckpt/best_model.ckpt ^
  --splits inference --nested --horizons 12000 --per_flight

:: Same, all three horizons (30 s / 60 s / 120 s) from one pass
python -m tools.eval_vel_horizons --config configs/exp/UAV/tilt_aware.conf ^
  --ckpt experiments/UAV/tilt_aware/ckpt/best_model.ckpt ^
  --splits inference --nested --horizons 3000 6000 12000 --per_flight

:: Same, and also save one row per window to a CSV
python -m tools.eval_vel_horizons --config configs/exp/UAV/tilt_aware.conf ^
  --ckpt experiments/UAV/tilt_aware/ckpt/best_model.ckpt ^
  --splits inference --nested --horizons 3000 6000 12000 --per_flight ^
  --win_csv tilt_aware_windows.csv


:: ---------------------------------------------------------------- ONE FLIGHT

:: ---------------------------------------------------------------- CONTROL
:: Is the network better than 3 plain numbers?  Fits one fixed accel
:: correction on train (no network, no training) and scores it next to
:: the trained model on the held-out flights.  "const" vs "net" rows.
python -m tools.const_bias_control --config configs/exp/UAV/tilt_aware.conf ^
  --ckpt experiments/UAV/tilt_aware/ckpt/best_model.ckpt ^
  --fit_split train --score_split inference --horizons 3000 6000 12000

:: Same, fitted at 120 s (best for long outages)
python -m tools.const_bias_control --config configs/exp/UAV/tilt_aware.conf ^
  --fit_split train --score_split inference --horizons 12000 --fit_horizon 12000


:: ---------------------------------------------------------------- CHECKS

:: ---------------------------------------------------------------- ONNX export
:: needs:  pip install onnx onnxruntime

  --ckpt experiments/UAV/tilt_rotate/ckpt/best_model.ckpt ^

:: ---------------------------------------------------------------- ONNX inference
:: Flight logs -> ONNX model -> vel/dir/pos errors, same numbers as eval_vel_horizons.

python -m tools.onnx_inference --onnx tilt_rotate_40s.onnx ^
  --config configs/exp/UAV/tilt_rotate.conf --splits inference ^
  --horizons 3000 4000 --out_csv onnx_windows.csv

:: one flight, and check ONNX against the PyTorch checkpoint on real data
python -m tools.onnx_inference --onnx tilt_rotate_40s.onnx ^
  --config configs/exp/UAV/tilt_rotate.conf --splits inference ^
  --csv 2026_02_06_143_23_sensor_data.csv --horizons 3000 4000 ^
  --compare_ckpt experiments/UAV/tilt_rotate/ckpt/best_model.ckpt

:: arrays only (no dataset code, no PyTorch): acc, gyro, g_body of (4009, 3)
